     1→/**
     2→ * Gulf Seismic Authority Graph — Master Data Layer
     3→ * --------------------------------------------------
     4→ * Single source of truth for the Country → City → Service → Industry →
     5→ * Project → Case Study → Blog authority graph.
     6→ *
     7→ * In production these entities are served from the headless WordPress CMS
     8→ * (https://cms.gulfseismic.com/graphql) via the CPTs: Countries, Cities,
     9→ * Services, Industries, Projects, CaseStudies, Blogs, FAQs, Resources.
    10→ * This module provides resilient seed data so the frontend renders even
    11→ * when the CMS is unreachable, and acts as the TypeScript contract for
    12→ * the GraphQL response shapes.
    13→ */
    14→
    15→export type CountrySlug = "uae" | "saudi-arabia";
    16→
    17→export interface Country {
    18→  slug: CountrySlug;
    19→  name: string;
    20→  shortName: string;
    21→  code: string; // ISO 3166-1 alpha-2
    22→  flag: string; // emoji
    23→  heroHeading: string;
    24→  heroDescription: string;
    25→  seoTitle: string;
    26→  seoDescription: string;
    27→  dialCode: string;
    28→  whatsapp: string;
    29→  phone: string;
    30→}
    31→
    32→export interface City {
    33→  slug: string;
    34→  country: CountrySlug;
    35→  name: string;
    36→  region: string;
    37→  latitude: number;
    38→  longitude: number;
    39→  heroHeading: string;
    40→  heroDescription: string;
    41→  seoTitle: string;
    42→  seoDescription: string;
    43→  population: string;
    44→  highlights: string[];
    45→}
    46→
    47→export interface Service {
    48→  slug: string;
    49→  name: string;
    50→  tagline: string;
    51→  shortDescription: string;
    52→  longDescription: string;
    53→  icon: string; // lucide icon name
    54→  benefits: string[];
    55→  materials: string[];
    56→  equipment: string[];
    57→  industriesServed: string[]; // industry slugs
    58→  faqs: { question: string; answer: string }[];
    59→  heroHeading: string;
    60→  heroDescription: string;
    61→  seoTitle: string;
    62→  seoDescription: string;
    63→  process: { step: number; title: string; description: string }[];
    64→  specs: { label: string; value: string }[];
    65→}
    66→
    67→export interface Industry {
    68→  slug: string;
    69→  name: string;
    70→  description: string;
    71→  icon: string;
    72→  challenges: string[];
    73→  solutions: string[];
    74→  services: string[]; // service slugs
    75→}
    76→
    77→export interface Project {
    78→  slug: string;
    79→  title: string;
    80→  country: CountrySlug;
    81→  city: string; // city slug
    82→  service: string; // service slug
    83→  industry: string; // industry slug
    84→  client: string;
    85→  year: number;
    86→  duration: string;
    87→  challenge: string;
    88→  solution: string;
    89→  execution: string;
    90→  materials: string[];
    91→  equipment: string[];
    92→  results: { label: string; value: string }[];
    93→  gallery: { alt: string; caption: string }[];
    94→  location: string;
    95→  area: string;
    96→}
    97→
    98→export interface CaseStudy {
    99→  slug: string;
   100→  title: string;
   101→  projectSlug: string;
   102→  summary: string;
   103→  outcomes: string[];
   104→  testimonial: { quote: string; author: string; role: string } | null;
   105→}
   106→
   107→export interface BlogPost {
   108→  slug: string;
   109→  title: string;
   110→  excerpt: string;
   111→  category: string;
   112→  readTime: string;
   113→  date: string;
   114→  author: string;
   115→}
